const hexHi1 = '0000';
const hexLow1 = '03E5';
const hexHi2 = '0000';
const hexLow2 = '03E1';
function hexHiLowToDecimal(hexHi,hexLow) {
    const combinedHex = hexHi + hexLow;
    const decimalValue = parseInt(combinedHex,16);
    return decimalValue;
}
